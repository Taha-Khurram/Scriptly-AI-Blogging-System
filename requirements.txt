# Pinned to the versions in the working local .venv (Python 3.11).
#
# Fully unpinned deps caused pip's resolver to backtrack through hundreds of
# transitive versions on Render and fail with "resolution-too-deep". Pinning
# both the direct deps and the shared Google/gRPC/HTTP transitive packages
# keeps the resolver's search space small and the build fast + reproducible.

# --- Web framework ---
flask==3.1.2
gunicorn==25.1.0
waitress==3.0.2
whitenoise==6.11.0
flask-compress==1.23
python-dotenv==1.2.1

# --- Security ---
# CSRF tokens for every state-changing request. SameSite=Lax covers the common
# cross-site form POST, but it is one browser-dependent control and does not
# cover same-site subdomain attacks.
flask-wtf==1.2.2
# Per-IP / per-user rate limiting. The public site exposes endpoints that spend
# a Gemini call each with no authentication in front of them.
flask-limiter==4.1.1
# HTML sanitisation for stored content rendered with autoescaping disabled.
# The [css] extra pulls tinycss2, needed to allowlist inline style properties
# rather than dropping the style attribute wholesale.
bleach[css]==6.2.0

# --- Data / infrastructure ---
firebase-admin==7.1.0
# Shared cache, rate-limit counters and the scheduler's single-runner lease.
# Optional at runtime: without REDIS_URL the app falls back to per-process
# state, which is correct for one worker only.
redis==6.4.0
